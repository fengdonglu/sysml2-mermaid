  `perform`.

```ts
private captureStatement(): string {
  const parts: string[] = [];
  let prev: Token | null = null;
  const isWord = (t: Token): boolean => t.type === 'ident' || t.type === 'number' || t.type === 'blockcomment';
  while (this.peek().type !== 'eof' && !this.at(';') && !this.at('}')) {
    const t = this.next();
    if (prev && isWord(prev) && isWord(t)) parts.push(' ');
    parts.push(t.type === 'blockcomment' ? `/* ${t.value} */` : t.value);
    prev = t;
  }
  this.eat(';');
  return parts.join('');
}
```

```ts
private parseFlow(): void {
  const kw = this.next();
  let payload = '';
  if (this.eat('of')) payload = this.parseReference();
  if (!this.expect('from')) return;
  const sourceRef = this.parseReference();
  if (!this.expect('to')) return;
  const targetRef = this.parseReference();
  this.expect(';');
  this.addRelationship('flow', sourceRef, targetRef, kw, payload ? { label: payload } : {});
}

private parsePerform(ownerId: string | undefined): void {
  const kw = this.next();
  this.eat('action');
  const targetRef = this.parseReference();
  this.expect(';');
  if (ownerId) this.addRelationship('perform', ownerId, targetRef, kw);
}

private parseParam(dir: string, ownerId: string | undefined, ownerQName: string): void {
  this.next();
  if (this.peek().type !== 'ident') { this.error('expected-name', 'expected parameter name'); return; }
  const el = this.makeElement('attribute', this.next(), ownerId, ownerQName);
  if (this.eat(':')) el.typeRef = this.parseQualifiedName();
  el.name = `${dir} ${el.name}`;
  this.expect(';');
}

private parseRawStatement(keyword: string, ownerId: string | undefined, ownerQName: string): void {
  const kw = this.next();
  const raw = this.captureStatement();
  const el = this.makeChild('action', ownerId, ownerQName, keyword, raw);
  this.prevRef = el.qualifiedName;
  void kw;
}
```

```ts
import type { Element, SysmlModel } from '../core/model/types.js';
import type { ViewDescription, ViewEdge, ViewNode } from './types.js';

const STATE_KINDS = new Set<Element['kind']>(['state-def', 'state']);
const INITIAL = new Set(['initial', 'start']);
const FINAL = new Set(['done', 'terminate']);

function rowsFor(model: SysmlModel, el: Element): string[] {
  const rows: string[] = [];
  for (const id of el.childIds) {
    const c = model.elements.get(id);
    if (!c) continue;
    if (c.kind === 'action') rows.push(`${c.name}: ${c.typeRef ?? ''}`.trim());
    else if (c.kind === 'state') rows.push(c.name);
  }
  return rows;
}

export function statemachineView(model: SysmlModel): ViewDescription {
  const nodes: ViewNode[] = [];
  const ids = new Set<string>();
  const add = (n: ViewNode) => { if (!ids.has(n.id)) { ids.add(n.id); nodes.push(n); } };
  const addElement = (el: Element) => add({ id: el.id, kind: el.kind, label: el.name, stereotype: el.kind === 'unresolved' ? 'unresolved' : 'state', rows: rowsFor(model, el), ports: [] });

  for (const el of model.elements.values()) if (STATE_KINDS.has(el.kind)) addElement(el);

  const edges: ViewEdge[] = [];
  let n = 0;
  for (const rel of model.relationships) {
    if (rel.kind !== 'transition') continue;
    const s = pseudoEndpoint(model, rel.sourceRef, rel.sourceId, 'initial');
    const t = pseudoEndpoint(model, rel.targetRef, rel.targetId, 'final');
    if (s.el) addElement(s.el); else add(s.node!);
    if (t.el) addElement(t.el); else add(t.node!);
    const source = s.el ? s.el.id : s.node!.id;
    const target = t.el ? t.el.id : t.node!.id;
    edges.push({ id: rel.id ?? `t:${n++}`, kind: 'transition', source, target, label: rel.label });
  }
  return { nodes, edges };
}

function pseudoEndpoint(model: SysmlModel, ref: string, id: string | undefined, stereotype: 'initial' | 'final') {
  const simple = ref.split(/[.:]/).pop() ?? ref;
  const pseudoSet = stereotype === 'initial' ? INITIAL : FINAL;
  if (id && model.elements.has(id)) return { el: model.elements.get(id)! };
  if (pseudoSet.has(simple)) {
    const key = `${stereotype}:${ref}`;
    return { node: { id: key, kind: 'control' as const, label: simple, stereotype, rows: [], ports: [] } as ViewNode };
  }
  return { node: { id: `unresolved:${ref}`, kind: 'unresolved' as const, label: simple, stereotype: 'unresolved', rows: [], ports: [] } as ViewNode };
}
```

```ts
import type { Element, SysmlModel } from '../core/model/types.js';
import type { ViewDescription, ViewEdge, ViewGroup, ViewNode } from './types.js';

const ACTION_KINDS = new Set<Element['kind']>(['action-def', 'action', 'control']);

function rowsFor(model: SysmlModel, el: Element): string[] {
  return el.childIds
    .map((id) => model.elements.get(id))
    .filter((c): c is Element => c?.kind === 'attribute')
    .map((c) => `${c.name}${c.typeRef ? ` : ${c.typeRef}` : ''}`);
}

export function activityView(model: SysmlModel): ViewDescription {
  const nodes: ViewNode[] = [];
  const ids = new Set<string>();
  const add = (n: ViewNode) => { if (!ids.has(n.id)) { ids.add(n.id); nodes.push(n); } };
  const addElement = (el: Element) => add({ id: el.id, kind: el.kind, label: el.name, stereotype: el.kind === 'unresolved' ? 'unresolved' : 'action', rows: rowsFor(model, el), ports: [] });
  const addRef = (ref: string, id: string | undefined, stereotype: 'initial' | 'control') => {
    if (id && model.elements.has(id)) { addElement(model.elements.get(id)!); return id; }
    const key = id && !id.startsWith('unresolved:') ? id : `${stereotype}:${ref.split(/[.:]/).pop()}`;
    add({ id: key, kind: 'control', label: ref.split(/[.:]/).pop()!, stereotype, rows: [], ports: [] });
    return key;
  };

  for (const el of model.elements.values()) if (ACTION_KINDS.has(el.kind)) addElement(el);

  const edges: ViewEdge[] = [];
  let n = 0;
  for (const rel of model.relationships) {
    if (rel.kind !== 'succession' && rel.kind !== 'flow') continue;
    const source = addRef(rel.sourceRef, rel.sourceId, rel.sourceRef.split(/[.:]/).pop() === 'start' ? 'initial' : 'control');
    const target = addRef(rel.targetRef, rel.targetId, 'control');
    edges.push({ id: rel.id ?? `e:${n++}`, kind: rel.kind, source, target, label: rel.label });
  }

  const groups: ViewGroup[] = [];
  const performerOf = new Map<string, string>();
  for (const rel of model.relationships) {
    if (rel.kind !== 'perform' || !rel.sourceId || !rel.targetId) continue;
    if (!performerOf.has(rel.targetId)) performerOf.set(rel.targetId, rel.sourceId);
  }
  if (performerOf.size > 0) {
    const byPerformer = new Map<string, string[]>();
    for (const [actionId, performerId] of performerOf) {
      const arr = byPerformer.get(performerId) ?? [];
      arr.push(actionId);
      byPerformer.set(performerId, arr);
    }
    for (const [performerId, members] of byPerformer) {
      const el = model.elements.get(performerId);
      groups.push({ id: `lane:${performerId}`, label: el?.name ?? performerId, members });
    }
  }

  return { nodes, edges, ...(groups.length ? { groups } : {}) };
}
```
